class YoutubeChannel
{
    List<Subscriber> subscribers

    addSubscriber(subscriber)
    removeSubscriber(subscriber)

    upload(video)
    {
        print("Uploading Video")
        notifySubscribers(video)
    }

    notifySubscribers(video)
    {
        for each subscriber in subscribers
        {
            subscriber.update(video)
        }
    }
}
